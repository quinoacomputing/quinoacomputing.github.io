136  //! \brief Update coefficients using constant coefficients for b' and kappa'
137  //! \details This where the mix mass-fraction beta SDE is made consistent
138  //! with the no-mix and fully mixed limits by specifying the SDE
139  //! coefficients, b and kappa as functions of b' and kappa'. We leave S
140  //! unchanged.
141  void update(
142  char depvar,
143  ncomp_t ncomp,
144  const std::map< tk::ctr::Product, tk::real >& moments,
145  const std::vector< kw::sde_bprime::info::expect::type >& bprime,
146  const std::vector< kw::sde_kappaprime::info::expect::type >& kprime,
147  const std::vector< kw::sde_rho2::info::expect::type >& rho2,
148  const std::vector< kw::sde_r::info::expect::type >& r,
149  std::vector< kw::sde_b::info::expect::type >& b,
150  std::vector< kw::sde_kappa::info::expect::type >& k,
151  std::vector< kw::sde_S::info::expect::type >& S ) const
152  {
153  for (ncomp_t c=0; c<ncomp; ++c) {
154  tk::real m = tk::ctr::lookup( tk::ctr::mean(depvar,c), moments );
155  tk::real v = tk::ctr::lookup( tk::ctr::variance(depvar,c), moments );
156 
157  if (m<1.0e-8 || m>1.0-1.0e-8) m = 0.5;
158  if (v<1.0e-8 && v>1.0-1.0e-8) v = 0.5;
159 
160  b[c] = bprime[c] * (1.0 - v / m / ( 1.0 - m ));
161  k[c] = kprime[c] * v;
162  }
163  }

217  //! \brief Update coefficients b', kappa', and S
218  //! \details This where the mix mass-fraction beta SDE is made consistent
219  //! with the no-mix and fully mixed limits by specifying the SDE
220  //! coefficients, b and kappa as functions of b' and kappa'. We also
221  //! specify S to force d<rho>/dt = 0, where <rho> = rho_2/(1+rY).
222  void update(
223  char depvar,
224  ncomp_t ncomp,
225  const std::map< tk::ctr::Product, tk::real >& moments,
226  const std::vector< kw::sde_bprime::info::expect::type >& bprime,
227  const std::vector< kw::sde_kappaprime::info::expect::type >& kprime,
228  const std::vector< kw::sde_rho2::info::expect::type >& rho2,
229  const std::vector< kw::sde_r::info::expect::type >& r,
230  std::vector< kw::sde_b::info::expect::type >& b,
231  std::vector< kw::sde_kappa::info::expect::type >& k,
232  std::vector< kw::sde_S::info::expect::type >& S ) const
233  {
234  using tk::ctr::lookup;
235  using tk::ctr::mean;
236  using tk::ctr::variance;
237  using tk::ctr::cen3;
238  // statistics nomenclature:
239  // Y = instantaneous mass fraction,
240  // R = instantaneous density,
241  // y = Y - <Y>, mass fraction fluctuation about its mean,
242  // r = R - <R>, density fluctuation about its mean,
243  // <Y> = mean mass fraction,
244  // <R> = mean density,
245  for (ncomp_t c=0; c<ncomp; ++c) {
246  tk::real m = lookup( mean(depvar,c), moments ); // <Y>
247  tk::real v = lookup( variance(depvar,c), moments ); // <y^2>
248  tk::real d = lookup( mean(depvar,c+ncomp), moments ); // <R>
249  tk::real d2 = lookup( variance(depvar,c+ncomp), moments ); // <r^2>
250  tk::real d3 = lookup( cen3(depvar,c+ncomp), moments ); // <r^3>
251 
252  if (m<1.0e-8 || m>1.0-1.0e-8) m = 0.5;
253  if (v<1.0e-8 && v>1.0-1.0e-8) v = 0.5;
254  b[c] = bprime[c] * (1.0 - v/m/(1.0-m));
255  k[c] = kprime[c] * v;
256  //b[c] = 1.0;
257  //k[c] = 0.5*v/(m*(1.0-m));
258 
259  if (d < 1.0e-8) {
260  std::cout << "d:" << d << " ";
261  d = 0.5;
262  }
263  tk::real R = 1.0 + d2/d/d;
264  tk::real B = -1.0/r[c]/r[c];
265  tk::real C = (2.0+r[c])/r[c]/r[c];
266  tk::real D = -(1.0+r[c])/r[c]/r[c];
267  tk::real diff =
268  B*d/rho2[c] +
269  C*d*d*R/rho2[c]/rho2[c] +
270  D*d*d*d*(1.0 + 3.0*d2/d/d + d3/d/d/d)/rho2[c]/rho2[c]/rho2[c];
271  S[c] = (rho2[c]/d/R +
272  2.0*k[c]/b[c]*rho2[c]*rho2[c]/d/d*r[c]*r[c]/R*diff - 1.0) / r[c];
273  if (S[c] < 0.0 || S[c] > 1.0) {
274  std::cout << S[c] << " ";
275  S[c] = 0.5;
276  }
277  }
278  }
279 };
