57 //! Charm++ message type for sending a string of strings separated by ';'.
58 //! \author J. Bakosi
59 struct StringsMsg : public CMessage_StringsMsg {
60  using value_type = std::vector< std::string >;
61  //! Constructor
62  explicit StringsMsg( const std::string& v )
63  { strncpy( strings, v.c_str(), CSTYLE_STRLEN ); }
64  //! Value accessor splitting the migrated C-style string at semi-colons
65  value_type get() {
66  std::string s( strings );
67  boost::char_separator<char> sep(";");
68  boost::tokenizer< boost::char_separator<char> > tokens( s, sep );
69  value_type v;
70  for (const auto& t : tokens) v.push_back( t );
71  return v;
72  }
73  //! Data

77 //! Charm++ message type for sending a vector of strings of strings separated by
78 //! semi-colons. The size of the vector must be determined at compile-time and
79 //! comes in via the template argument. This avoids using variable-size
80 //! messages, which is more complicated, more error-prone, and most likely less
81 //! amenable to optimizations. The constructor and the value accessor interface
82 //! with std::vector< std::string >, but converted to POD so that the Charm++
83 //! runtime system easily serialize the data and migrate it over the network.
84 //! \author J. Bakosi
85 template< std::size_t Size = 1 >
86 struct VecStrsMsg : public CMessage_VecStrsMsg< Size > {
87  using value_type = std::vector< std::vector< std::string > >;
88  //! Constructor
89  explicit VecStrsMsg( const std::vector< std::string >& v ) {
90  Assert( Size <= v.size(), "Size of input vector to VecStrsMsg is larger "
91  "than specified in template argument" );
92  for (std::size_t i=0; i<Size; ++i)
93  strncpy( strings[i], v[i].c_str(), CSTYLE_STRLEN );
94  }
95  //! Value accessor
96  value_type get() {
97  value_type v( Size );
98  for (std::size_t i=0; i<Size; ++i) {
99  std::string s( strings[i] );
100  boost::char_separator<char> sep(";");
101  boost::tokenizer< boost::char_separator<char> > tokens( s, sep );
102  for (const auto& t : tokens) v[i].push_back( t );
103  }
104  return v;
105  }
106  //! Data
107  char strings[Size][CSTYLE_STRLEN];
108 };
