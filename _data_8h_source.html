100  //! Const ptr to physical variable access dispatch
101  //! \details Public interface to the first half of a physical variable
102  //! access. cptr() and var() are two member functions intended to be used
103  //! together in case when component and offset would be expensive to
104  //! compute for data access via the function call operator, i.e., cptr()
105  //! can be used to pre-compute part of the address, which returns a
106  //! pointer and var() can be used to finish the data access using the
107  //! pointer returned by cptr(). In other words, cptr() returns part of the
108  //! address known based on component and offset and intended to be used in
109  //! a setup phase. Then var() takes this partial address and finishes the
110  //! address calculation given the unknown id. Thus the following two data
111  //! accesses are equivalent (modulo constness):
112  //! * real& value = operator()( unk, comp, offs ); and
113  //! * const real* p = cptr( comp, offs ); and
114  //! const real& value = var( p, unk ); or real& value = var( p, unk );
115  //! Requirement: offset + component < nprop, enforced with an assert in
116  //! DEBUG mode, see also the constructor.
117  //! \param[in] component Component index, i.e., position of a scalar within
118  //! a system

120  //! equations among other systems

377  //! Remove a number of unknowns
378  //! \param[in] unknown Set of indices of unknowns to remove
379  //! \author J. Bakosi
380  void rm( const std::set< ncomp_t >& unknown ) {
381 
382  auto rem = [ &unknown ]( std::size_t i ) -> bool {
383  if (unknown.find(i) != end(unknown)) return true;
384  return false;
385  };
386 
387  std::size_t last = 0;
388  for(std::size_t i=0; i<m_nunk; ++i, ++last) {
389  while( rem(i) ) ++i;
390  if (i >= m_nunk) break;
391  for (ncomp_t p = 0; p<m_nprop; ++p) m_vec[ last+p ] = m_vec[ i+p ];
392  }
393  m_vec.resize( last*m_nprop );
394  m_nunk -= unknown.size();
395  }

611 //! Compute the maximum difference between the elements of two Data objects
612 //! \param[in] lhs 1st Data object
613 //! \param[in] rhs 2nd Data object
614 //! \return The index, i.e., the raw position, of and the largest absolute value
615 //! of the difference between all corresponding elements of _lhs_ and _rhs_.
616 //! \details The position returned is the position in the underlying raw data
617 //! structure, independent of components, offsets, etc. If lhs == rhs with
618 //! precision std::numeric_limits< tk::real >::epsilon(), a pair of (0,0.0)
619 //! is returned.
620 //! \note The Data objects _lhs_ and _rhs_ must have the same number of
621 //! unknowns and properties.
622 template< uint8_t Layout >
623 std::pair< std::size_t, tk::real >
624 maxdiff( const Data< Layout >& lhs, const Data< Layout >& rhs ) {
625  Assert( lhs.nunk() == rhs.nunk(), "Number of unknowns unequal" );
626  Assert( lhs.nprop() == rhs.nprop(), "Number of properties unequal" );
627  auto l = lhs.data().cbegin();
628  auto r = rhs.data().cbegin();
629  std::pair< std::size_t, tk::real > m( 0, std::abs(*l - *r) );
630  ++l; ++r;
631  while (l != lhs.data().cend()) {
632  const auto d = std::abs(*l - *r);
633  if (d > m.second) m = { std::distance(lhs.data().cbegin(),l), d };
634  ++l; ++r;
635  }
636  return m;
637 }
