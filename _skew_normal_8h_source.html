2 /*!
3  \file src/DiffEq/SkewNormal.h
4  \author J. Bakosi
5  \date Sat 20 Jun 2015 11:36:02 AM MDT
6  \copyright 2012-2015, Jozsef Bakosi.
7  \brief System of skew-normal SDEs
8  \details This file implements the time integration of a system of stochastic
9  differential equations (SDEs), whose invariant is the joint [skew-normal
10  distribution](http://en.wikipedia.org/wiki/Skew_normal_distribution).
11 
12  In a nutshell, the equation integrated governs a set of scalars,
13  \f$x_\alpha\f$, \f$\alpha\!=\!1,\dots,N\f$, as
14  \f[
15  \mathrm{d}x_\alpha(t) = -\frac{1}{T_\alpha}\left[x_\alpha -
16  \lambda_\alpha\sigma^2_\alpha\sqrt{\frac{2}{\pi}} \cdot
17  \frac{\exp{\left(-\lambda_\alpha^2x^2_\alpha/2\right)}}{1+\mathrm{erf}
18  \left( \lambda_\alpha x_\alpha/\sqrt{2}\right)} \right] \mathrm{d}t +
19  \sqrt{\frac{2\sigma^2_\alpha}{T_\alpha}}\mathrm{d}W_\alpha(t).
20  \f]
21  The invariant distribution is the joint skew-normal distribution
22  \f[
23  p(x_\alpha) = \frac{1}{\sigma_\alpha\sqrt{2\pi}} \exp\left(
24  -\frac{x^2_\alpha}{2\sigma^2_\alpha} \right) \left[1 + \mathrm{erf}\left(
25  \frac{\lambda_\alpha x_\alpha}{\sqrt{2}}\right) \right].
26  \f]
27  Here \f$\mathrm{erf}(y) = 2/\sqrt{\pi} \int_0^y \exp(-u^2) \mathrm{d}u\f$,
28  \f$T_\alpha\f$ are time scales, \f$\sigma_\alpha\f$ are variance parameters,
29  \f$\mathrm{d}W_\alpha(t)\f$ is an isotropic [Wiener
30  process](http://en.wikipedia.org/wiki/Wiener_process) with independent
31  increments, and \f$\lambda_\alpha\f$ are the parameters that control the
32  asymmetry and skewness for variable \f$x_\alpha\f$: \f$\lambda_\alpha<0\f$
33  and \f$\lambda_\alpha>0\f$ give negative and positive skewness,
34  respectively, while \f$\lambda_\alpha=0\f$ reduces the system to a set of
35  independent Ornstein-Uhlenbeck processes. See also
36  DiffEq/DiagOrnsteinUhlenbeck.h. For more details on the skew-normal
37  distribution, see http://www.jstor.org/stable/2337278.
38 */

101  //! \brief Advance particles according to the system of skew-normal SDEs
102  //! \param[inout] particles Array of particle properties
103  //! \param[in] stream Thread (or more precisely stream) ID
104  //! \param[in] dt Time step size
105  //! \param[in] moments Map of statistical moments
106  //! \author J. Bakosi
107  void advance( tk::ParProps& particles,
108  int stream,
109  tk::real dt,
110  const std::map< tk::ctr::Product, tk::real >& moments )
111  {
112  const auto npar = particles.npar();
113  for (auto p=decltype(npar){0}; p<npar; ++p) {
114  // Generate Gaussian random numbers with zero mean and unit variance
115  tk::real dW[m_ncomp];
116  m_rng.gaussian( stream, m_ncomp, dW );
117 
118  // Advance all m_ncomp scalars
119  for (ncomp_t i=0; i<m_ncomp; ++i) {
120  tk::real& x = particles( p, i, m_offset );
121  tk::real d = 2.0 * m_sigmasq[i] / m_T[i] * dt;
122  d = (d > 0.0 ? std::sqrt(d) : 0.0);
123  x += - ( x - m_lambda[i] * m_sigmasq[i]
124  * std::sqrt( 2.0 / M_PI )
125  * std::exp( - m_lambda[i] * m_lambda[i] * x * x / 2.0 )
126  / ( 1.0 + std::erf( m_lambda[i] * x / std::sqrt(2.0) ) )
127  ) / m_T[i] * dt
128  + d*dW[i];
129  }
130  }
131  }
