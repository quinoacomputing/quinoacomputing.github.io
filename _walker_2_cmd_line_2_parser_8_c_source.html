66  // Create PEGTL string input from std::string (i.e. concatenated argv[])
67  pegtl::string_input< ctr::Location > input( m_string );
68 
69  // Create PEGTLCmdLine to store parsed command line data which derives from
70  // CmdLine and has location() used during parsing. Also pass in HelpFactory,
71  // ctrinfo, containing the input deck keywords.
72  cmd::PEGTLCmdLine cmd( input, g_inputdeck.get< tag::cmd, tag::ctrinfo >() );
73 
74  // Reset parser's output stream to that of print's. This is so that mild
75  // warnings emitted during parsing can be output using the pretty printer.
76  // Usually, errors and warnings are simply accumulated during parsing and
77  // printed during diagnostics after the parser has finished. However, in some
78  // special cases we can provide a more user-friendly message right during
79  // parsing since there is more information available to construct a more
80  // sensible message. This is done in e.g., tk::grm::store_option. Resetting
81  // the global g_print, to that of passed in as the constructor argument allows
82  // not to have to create a new pretty printer, but use the existing one.
83  tk::grm::g_print.reset( print.save() );
84 
85  // Parse command line string by populating the underlying tagged tuple:
86  // basic_parse() below gives debug info during parsing, use it for debugging
87  // the parser itself, i.e., when modifying the grammar, otherwise, use
88  // dummy_parse() to compile faster
89  pegtl::dummy_parse< cmd::read_string >( input, cmd );
90 
91  // Echo errors and warnings accumulated during parsing
92  diagnostics( print, cmd.get< tag::error >() );
93 
94  // Strip command line (and its underlying tagged tuple) from PEGTL instruments
95  // and transfer it out
96  cmdline = std::move( cmd );
97 
98  // If we got here, the parser succeeded
99  print.item( "Parsed command line", "success" );
100 
101  // Print out help on all command-line arguments if the executable was invoked
102  // without arguments or the help was requested
103  const auto helpcmd = cmdline.get< tag::help >();
104  if (argc == 1 || helpcmd)
105  print.help< tk::QUIET >( WALKER_EXECUTABLE, cmdline.get< tag::cmdinfo >(),
106  "Command-line Parameters:", "-" );
107 
108  // Print out help on all control file keywords if they were requested
109  const auto helpctr = cmdline.get< tag::helpctr >();
110  if (helpctr)
111  print.help< tk::QUIET >( WALKER_EXECUTABLE, cmdline.get< tag::ctrinfo >(),
112  "Control File Keywords:" );
113 
114  // Print out verbose help for a single keyword if requested
115  const auto helpkw = cmdline.get< tag::helpkw >();
116  if (!helpkw.keyword.empty())
117  print.helpkw< tk::QUIET >( WALKER_EXECUTABLE, helpkw );
118 
119  // Immediately exit if any help was output or was called without any argument
120  if (argc == 1 || helpcmd || helpctr || !helpkw.keyword.empty()) CkExit();
121 
122  // Make sure mandatory arguments are set
123  auto alias = kw::control().alias();

125  "Mandatory control file not specified. "
126  "Use '--" + kw::control().string() + " <filename>'" +

128 }
