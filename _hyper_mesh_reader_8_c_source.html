45 //! \return Vector of strings containing the filenames

48 {
49  // Read XML file for metadata
50  pugi::xml_document meta;
51  auto res = meta.load_file( m_filename.c_str() );
52  ErrChk( res == true, "Could not parse XML attempting to read HyperMesh "
53  "metadata file '" + m_filename + "', at character position " +
54  std::to_string(res.offset) + " : " + res.description() );
55 
56  // Extract path from metadata filename
57  std::string fn = m_filename;
58  std::string path( fn.substr(0,fn.find_last_of("/")+1) );
59  // Prepend path to filenames to be extracted
60  std::pair< std::string, std::string > filenames( path, path );
61 
62  for (const auto& child : meta.children("mesh"))
63  for (const auto& group : child.children()) {
64 
65  if (group.name() == std::string("coordinates")) {
66 
67  filenames.first += group.attribute("file").value();
68 
69  } else if (group.name() == std::string("element_set")) {
70 
71  filenames.second += group.attribute("file").value();
72  Assert( group.attribute("topology").value() ==
73  std::string("four_node_tet"), "Only pure tetrahedron-element "
74  "meshes are supported" );
75  }
76  }

82 HyperMeshReader::readNodes( const std::string& filename, UnsMesh& mesh ) const
83 // *****************************************************************************
84 // Read nodes
85 //! \param[in] filename Filename to read nodes from
86 //! \param[in] mesh Unstructured mesh object to put nodes coordinates
87 //! \note We throw away the node ID, which means the nodes must be in order.
88 //! \author J. Bakosi
89 // *****************************************************************************
90 {
91  // Read in node coordinates: x-coord y-coord z-coord
92  for (auto& line : tk::Reader(filename).lines()) {
93  std::stringstream ss(line);
94  int id;
95  tk::real x, y, z;
96  ss >> id >> x >> y >> z;
97  mesh.x().push_back( x );
98  mesh.y().push_back( y );
99  mesh.z().push_back( z );
100  }
101 }
