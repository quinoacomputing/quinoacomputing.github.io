69 std::map< tk::ctr::RawRNGType, tk::RNG >
70 RNGStack::selected( const std::vector< tk::ctr::RNGType >& sel ) const
71 // *****************************************************************************
72 // Instantiate selected RNGs from factory and place them in map
73 //! \param[in] sel Vector of selected RNGs to instantiate (selected by the user)
74 //! \return A std::map of keys and their associated instantiated RNG objects
75 //! \author J. Bakosi
76 // *****************************************************************************
77 {
78  using tk::ctr::RawRNGType;
79  std::map< RawRNGType, tk::RNG > rng;
80  for (const auto& s : sel) {
81  const auto r = m_factory.find(s);
82  if (r != end(m_factory))
83  rng.emplace( static_cast< RawRNGType >( s ), r->second() );
84  else Throw( "RNG not found in factory" );
85  }
86  return rng;
87 }
