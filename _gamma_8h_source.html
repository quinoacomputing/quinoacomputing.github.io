4  \author J. Bakosi
5  \date Wed 04 May 2016 11:20:58 AM MDT
6  \copyright 2012-2015, Jozsef Bakosi, 2016, Los Alamos National Security, LLC.
7  \brief System of gamma SDEs
8  \details This file implements the time integration of a system of stochastic
9  differential equations (SDEs), with linear drift and linear diagonal
10  diffusion, whose invariant is the joint [gamma
11  distribution](http://en.wikipedia.org/wiki/Gamma_distribution).
12 
13  In a nutshell, the equation integrated governs a set of scalars,
14  \f$0\!\le\!Y_\alpha\f$, \f$\alpha\!=\!1,\dots,N\f$, as
15  \f[
16  \mathrm{d}Y_\alpha(t) = \frac{b_\alpha}{2}\big[S_\alpha -
17  (1-S_\alpha)Y_\alpha\big]\mathrm{d}t + \sqrt{\kappa_\alpha Y_\alpha}
18  \mathrm{d}W_\alpha(t), \qquad \alpha=1,\dots,N
19  \f]
20  with parameter vectors \f$b_\alpha > 0\f$, \f$\kappa_\alpha > 0\f$, and \f$0
21  < S_\alpha < 1\f$. Here \f$\mathrm{d}W_\alpha(t)\f$ is an isotropic
22  vector-valued [Wiener process](http://en.wikipedia.org/wiki/Wiener_process)
23  with independent increments. The invariant distribution is the joint gamma
24  distribution. This system of SDEs consists of N independent equations.
25 */

90  //! \brief Advance particles according to the system of gamma SDEs
91  //! \param[in,out] particles Array of particle properties
92  //! \param[in] stream Thread (or more precisely stream) ID
93  //! \param[in] dt Time step size
94  //! \author J. Bakosi
95  void advance( tk::Particles& particles,
96  int stream,
97  tk::real dt,
98  tk::real,
99  const std::map< tk::ctr::Product, tk::real >& )
100  {
101  const auto npar = particles.nunk();
102  for (auto p=decltype(npar){0}; p<npar; ++p) {
103  // Generate Gaussian random numbers with zero mean and unit variance
104  std::vector< tk::real > dW( m_ncomp );
105  m_rng.gaussian( stream, m_ncomp, dW.data() );
106 
107  // Advance all m_ncomp scalars
108  for (ncomp_t i=0; i<m_ncomp; ++i) {
109  tk::real& par = particles( p, i, m_offset );
110  tk::real d = m_k[i] * par * dt;
111  d = (d > 0.0 ? std::sqrt(d) : 0.0);
112  par += 0.5*m_b[i]*(m_S[i] - (1.0 - m_S[i])*par)*dt + d*dW[i];
113  }
114  }
115  }
