238  //! \brief Parser error and warning message handler.
239  //! \details This function is used to associated and dispatch an error or a
240  //! warning during parsing. After finding the error message corresponding to
241  //! a key, it pushes back the message to a std::vector of std::string, which
242  //! then will be diagnosed later by tk::FileParser::diagnostics. The
243  //! template arguments define (1) the grammar stack (Stack, a tagged tuple)
244  //! to operate on, (2) the message type (error or warning), and (3) the
245  //! message key used to look up the error message associated with the key.
246  //! \param[in,out] stack Grammar stack (a tagged tuple) to operate on
247  //! \param[in] value Last parsed token (can be empty, depending on what
248  //! context this function gets called.
249  //! \author J. Bakosi
250  template< class Stack, MsgType type, MsgKey key >
251  static void Message( Stack& stack, const std::string& value ) {
252  const auto& msg = message.find(key);
253  if (msg != message.end()) {
254  std::stringstream ss;
255  const std::string typestr( type == MsgType::ERROR ? "Error" : "Warning" );
256  if (!value.empty()) {
257  ss << typestr << " while parsing '" << value << "' at "
258  << stack.location() << ". " << msg->second;
259  } else {
260  ss << typestr << " while parsing at " << stack.location() << ". "
261  << msg->second;
262  }
263  stack.template push_back< tag::error >( ss.str() );
264  } else {
265  stack.template push_back< tag::error >
266  ( std::string("Unknown parser ") +
267  (type == MsgType::ERROR ? "error" : "warning" ) +
268  " with no location information." );
269  }
270  }

304  //! \brief Put option (i.e., a tk::Toggle) in grammar state (or stack) at a
305  //! position given by tags
306  //! \details This function is used to store an option (an object deriving from
307  //! tk::Toggle) into the grammar stack. See walker::ctr::DiffEq for an
308  //! example specialization of tk::Toggle to see how an option is created
309  //! from tk::Toggle.) The grammar stack is a hiearchical tagged tuple and
310  //! the variadic list of template arguments, tags..., are used to specify
311  //! a series tags (empty structs, see Control/Tags.h) addressing a
312  //! particular field of the tagged tuple, i.e., one tag for every additional
313  //! depth level.
314  //! \param[in,out] stack Grammar stack (a tagged tuple) to operate on
315  //! \param[in] value Last parsed token
316  //! \param[in] defaults Reference to a copy of the full grammar stack at the
317  //! initial state, i.e., containing the defaults for all of its fields. This
318  //! is used to detect if the user wants to overwrite an option value that
319  //! has already been set differently from the default
320  //! \author J. Bakosi
321  template< class Stack, template< class > class use, class Option,
322  class DefaultStack, class... tags >
323  static void store_option( Stack& stack,
324  const std::string& value,
325  const DefaultStack& defaults ) {
326  Option opt;
327  if (opt.exist(value)) {
328  // Emit warning on overwriting a non-default option value. This is
329  // slightly inelegant. To be more elegant, we could simply call Message()
330  // here, but the warning message can be more customized here (inside of
331  // this function) and thus produces a more user-friendly message, compared
332  // to a static message that Message() operates with due to its generic
333  // nature. Instead, we emit this more user-friendly message here
334  // (during parsing), instead of after parsing as part of the final parser-
335  // diagnostics. We still provide location information here though.
336  if (stack.template get< tags... >() != opt.value( value ) &&
337  stack.template get< tags... >() != defaults.template get< tags... >())
338  g_print << "\n>>> WARNING: Multiple definitions for '"
339  << opt.group() << "' option. Overwriting '"
340  << opt.name( stack.template get< tags... >() ) << "' with '"
341  << opt.name( opt.value( value ) ) << "' at "
342  << stack.location() << ".\n\n";
343  stack.template set< tags... >( opt.value( value ) );
344  } else {
345  Message< Stack, ERROR, MsgKey::NOOPTION >( stack, value );
346  }
347  // trigger error at compile-time if any of the expected option values
348  // is not in the keywords pool of the grammar
349  boost::mpl::for_each< typename Option::keywords >( is_keyword< use >() );
350  }

1508  //! pre-defined keywords
1509  //! \details In grammar definitions, every keyword should be wrapped around
1510  //! this use template, which conditionally inherits the keyword type its
1511  //! templated on (as defined in Control/Keywords.h) if the keyword is listed

1513  //! argument and zero or more additional pools. If the keyword is not in any
1514  //! of the pools, a compiler error is generated, since the struct cannot
1515  //! inherit from base class 'char'. In that case, simply add the new keyword
1516  //! into one of the pools of keywords corresponding to the given grammar.
1517  //! The rationale behind this wrapper is to force the developer to maintain
1518  //! the keywords pool for a grammar. The pools are boost::mpl::sets and are
1519  //! used to provide help on command line arguments for a given executable.
1520  //! They allow compile-time iteration with boost::mpl::for_each or
1521  //! generating a run-time std::map associating, e.g., keywords to their help
1522  //! strings.
1523  //! \warning Since the default maximum number of elements in a
1524  //! boost::mpl::set is 20, and increasing this number would require custom
1525  //! boost::preprocessor-generated boost::mpl headers, which is cumbersome
1526  //! and error-prone, instead we work with several pools here that can each
1527  //! hold a maximum 20 items and use boost::mpl::or_ and boost::mpl::has_key
1528  //! to search for the keyword in either of the pools. Note that
1529  //! boost::mpl::or_ allows a maximum 5 template arguments by definition,

1532  //! BOOST_MPL_LIMIT_METAFUNCTION_ARITY at the top of this file.
1533  //! \warning Note that an even more elegant solution to the problem this
1534  //! wrapper is intended to solve is to use a metaprogram that collects all
1535  //! occurrences of the keywords in a grammar. However, that does not seem to
1536  //! be possible without extensive macro-trickery. Instead, if all keywords
1537  //! in all grammar definitions are wrapped inside this use template (or one
1538  //! of its specializations), we make sure that only those keywords can be
1539  //! used by a grammar that are listed in the pool corresponding to a
1540  //! grammar. However, this is still only a partial solution, since listing
1541  //! more keywords in the pool than those used in the grammar is still
1542  //! possible, which would result in those keywords included in, e.g., the
1543  //! on-screen help generated even though some of the keywords may not be
1544  //! implemented by the given grammar. So please don't abuse and don't list
1545  //! keywords in the pool only if they are implemented in the grammar.
1546  //! \see For example usage with a single pool, see the template typedef
1547  //! walker::cmd::use in Control/Walker/CmdLine/Grammar.h and its keywords
