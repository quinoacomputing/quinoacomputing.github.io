2 /*!
3  \file src/DiffEq/Dirichlet.h
4  \author J. Bakosi
5  \date Wed 04 May 2016 11:13:01 AM MDT
6  \copyright 2012-2015, Jozsef Bakosi, 2016, Los Alamos National Security, LLC.
7  \brief Dirichlet SDE
8  \details This file implements the time integration of a system of stochastic
9  differential equations (SDEs), whose invariant is the [Dirichlet
10  distribution](http://en.wikipedia.org/wiki/Dirichlet_distribution).
11 
12  In a nutshell, the equation integrated governs a set of scalars,
13  \f$0\!\le\!Y_\alpha\f$, \f$\alpha\!=\!1,\dots,N-1\f$,
14  \f$\sum_{\alpha=1}^{N-1}Y_\alpha\!\le\!1\f$, as
15  \f[
16  \mathrm{d}Y_\alpha(t) = \frac{b_\alpha}{2}\big[S_\alpha Y_N -
17  (1-S_\alpha)Y_\alpha\big]\mathrm{d}t + \sqrt{\kappa_\alpha Y_\alpha
18  Y_N}\mathrm{d}W_\alpha(t), \qquad \alpha=1,\dots,N-1
19  \f]
20  with parameter vectors \f$b_\alpha > 0\f$, \f$\kappa_\alpha > 0\f$, and \f$0
21  < S_\alpha < 1\f$, and \f$Y_N = 1-\sum_{\alpha=1}^{N-1}Y_\alpha\f$. Here
22  \f$\mathrm{d}W_\alpha(t)\f$ is an isotropic vector-valued [Wiener
23  process](http://en.wikipedia.org/wiki/Wiener_process) with independent
24  increments. The invariant distribution is the Dirichlet distribution,
25  provided the parameters of the drift and diffusion terms satisfy
26  \f[
27  (1-S_1) b_1 / \kappa_1 = \dots = (1-S_{N-1}) b_{N-1} / \kappa_{N-1}.
28  \f]
29  To keep the invariant distribution Dirichlet, the above constraint on the
30  coefficients must be satisfied. For more details on the Dirichlet SDE, see
31  http://dx.doi.org/10.1155/2013/842981.
32 */

97  //! \brief Advance particles according to the Dirichlet SDE
98  //! \param[in,out] particles Array of particle properties
99  //! \param[in] stream Thread (or more precisely stream) ID
100  //! \param[in] dt Time step size
101  //! \author J. Bakosi
102  void advance( tk::Particles& particles,
103  int stream,
104  tk::real dt,
105  tk::real,
106  const std::map< tk::ctr::Product, tk::real >& )
107  {
108  const auto npar = particles.nunk();
109  for (auto p=decltype(npar){0}; p<npar; ++p) {
110  // Compute Nth scalar
111  tk::real yn = 1.0 - particles(p, 0, m_offset);
112  for (ncomp_t i=1; i<m_ncomp; ++i)
113  yn -= particles( p, i, m_offset );
114 
115  // Generate Gaussian random numbers with zero mean and unit variance
116  std::vector< tk::real > dW( m_ncomp );
117  m_rng.gaussian( stream, m_ncomp, dW.data() );
118 
119  // Advance first m_ncomp (K=N-1) scalars
120  for (ncomp_t i=0; i<m_ncomp; ++i) {
121  tk::real& par = particles( p, i, m_offset );
122  tk::real d = m_k[i] * par * yn * dt;
123  d = (d > 0.0 ? std::sqrt(d) : 0.0);
124  par += 0.5*m_b[i]*( m_S[i]*yn - (1.0-m_S[i]) * par )*dt + d*dW[i];
125  }
126  }
127  }
