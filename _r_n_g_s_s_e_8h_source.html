34  //! Adaptor to use a std distribution with the RNGSSE generator
35  //! \see C++ concepts: UniformRandomNumberGenerator

72  //! Uniform RNG: Generate uniform random numbers
73  //! \param[in] tid Thread (or more precisely) stream ID
74  //! \param[in] num Number of RNGs to generate
75  //! \param[in,out] r Pointer to memory to write the random numbers to
76  void uniform( int tid, ncomp_t num, double* r ) const {
77  for (ncomp_t i=0; i<num; ++i)
78  r[i] = static_cast<double>(
79  Generate( &m_stream[ static_cast<std::size_t>(tid) ] ) )
80  / 4294967296.0;
81  }

83  //! Gaussian RNG: Generate Gaussian random numbers
84  //! \param[in] tid Thread (or more precisely stream) ID
85  //! \param[in] num Number of RNGs to generate
86  //! \param[in,out] r Pointer to memory to write the random numbers to
87  //! \details Generating Gaussian random numbers is implemented via an
88  //! adaptor, modeling std::UniformRandomNumberGenerator, outsourcing the
89  //! transformation of uniform random numbers to Gaussian ones, to the
90  //! standard library. The adaptor is instantiated here because a standard
91  //! distribution, such as e.g., std::normal_distribution, generates
92  //! numbers using operator() with no arguments, thus the RNG state and the
93  //! thread ID (this latter only known here) must be stored in the adaptor
94  //! functor's state. Even though creating the adaptor seems like a
95  //! potentially costly operation for every call, using the standard
96  //! library implementation is still faster than a hand-coded
97  //! implementation of the Box-Muller algorithm. Note that libc++ uses a
98  //! cache, as Box-Muller, implemented using the polar algorithm generates
99  //! 2 Gaussian numbers for each pair of uniform ones, caching every 2nd.
100  void gaussian( int tid, ncomp_t num, double* r ) const {
101  Adaptor generator( m_stream, tid );
102  std::normal_distribution<> gauss_dist( 0.0, 1.0 );
103  for (ncomp_t i=0; i<num; ++i) r[i] = gauss_dist( generator );
104  }

106  //! Beta RNG: Generate beta random numbers
107  //! \param[in] tid Thread (or more precisely stream) ID
108  //! \param[in] num Number of RNGs to generate
109  //! \param[in] p First beta shape parameter
110  //! \param[in] q Second beta shape parameter
111  //! \param[in] a Beta displacement parameter
112  //! \param[in] b Beta scale factor
113  //! \param[in,out] r Pointer to memory to write the random numbers to
114  //! \details Generating beta-distributed random numbers is implemented via
115  //! an adaptor, modeling boost::UniformRandomNumberGenerator, outsourcing
116  //! the transformation of uniform random numbers to beta-distributed ones,
117  //! to boost::random. The adaptor is instantiated here because a boost
118  //! random number distribution, such as e.g.,
119  //! boost::random::beta_distribution, generates numbers using operator()
120  //! with no arguments, thus the RNG state and the thread ID (this latter
121  //! only known here) must be stored in the adaptor functor's state.
122  void beta( int tid, ncomp_t num, double p, double q, double a, double b,
123  double* r ) const {
124  Adaptor generator( m_stream, tid );
125  boost::random::beta_distribution<> beta_dist( p, q );
126  for (ncomp_t i=0; i<num; ++i) r[i] = beta_dist( generator ) * b + a;
127  }
