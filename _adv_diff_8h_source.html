60  //! Compute the left hand side sparse matrix
61  //! \param[in] coord Mesh node coordinates
62  //! \param[in] inpoel Mesh element connectivity
63  //! \param[in] psup Linked lists storing IDs of points surrounding points
64  //! \param[in,out] lhsd Diagonal of the sparse matrix storing nonzeros
65  //! \param[in,out] lhso Off-diagonal of the sparse matrix storing nonzeros
66  //! \details Sparse matrix storing the nonzero matrix values at rows and
67  //! columns given by psup. The format is similar to compressed row
68  //! storage, but the diagonal and off-diagonal data are stored in separate
69  //! vectors. For the off-diagonal data the local row and column indices,
70  //! at which values are nonzero, are stored by psup (psup1 and psup2,
71  //! where psup2 holds the indices at which psup1 holds the point ids
72  //! surrounding points, see also tk::genPsup()). Note that the number of
73  //! mesh points (our chunk) npoin = psup.second.size()-1.
74  //! \author J. Bakosi
75  void lhs( const std::array< std::vector< tk::real >, 3 >& coord,
76  const std::vector< std::size_t >& inpoel,
77  const std::pair< std::vector< std::size_t >,
78  std::vector< std::size_t > >& psup,
79  tk::MeshNodes& lhsd,
80  tk::MeshNodes& lhso )
81  {
82  Assert( psup.second.size()-1 == coord[0].size(),
83  "Number of mesh points and number of global IDs unequal" );
84  Assert( lhsd.nunk() == psup.second.size()-1, "Number of unknowns in "
85  "diagonal sparse matrix storage incorrect" );
86  Assert( lhso.nunk() == psup.first.size(), "Number of unknowns in "
87  "off-diagonal sparse matrix storage incorrect" );
88 
89  // Lambda to compute the sparse matrix vector index for row and column
90  // indices. Used only for off-diagonal entries.
91  auto spidx = [ &psup ]( std::size_t r, std::size_t c ) -> std::size_t {
92  Assert( r != c, "Only for computing the off-diagonal indices" );
93  for (auto i=psup.second[r]+1; i<=psup.second[r+1]; ++i)
94  if (c == psup.first[i]) return i;
95  Throw( "Cannot find row, column: " + std::to_string(r) + ',' +
96  std::to_string(c) + " in sparse matrix" );
97  };

99  const auto& x = coord[0];
100  const auto& y = coord[1];
101  const auto& z = coord[2];
102 
103  // Zero matrix for all components
104  for (ncomp_t c=0; c<m_ncomp; ++c) {
105  lhsd.fill( c, m_offset, 0.0 );
106  lhso.fill( c, m_offset, 0.0 );
107  }
108 
109  for (std::size_t e=0; e<inpoel.size()/4; ++e) {
110  const auto A = inpoel[e*4+0];
111  const auto B = inpoel[e*4+1];
112  const auto C = inpoel[e*4+2];
113  const auto D = inpoel[e*4+3];
114  std::array< tk::real, 3 > ba{{ x[B]-x[A], y[B]-y[A], z[B]-z[A] }},
115  ca{{ x[C]-x[A], y[C]-y[A], z[C]-z[A] }},
116  da{{ x[D]-x[A], y[D]-y[A], z[D]-z[A] }};
117  const auto J = tk::triple( ba, ca, da ) / 120.0;
118 
119  for (ncomp_t c=0; c<m_ncomp; ++c) {
120  const auto r = lhsd.cptr( c, m_offset );
121  lhsd.var( r, A ) += 2.0 * J;
122  lhsd.var( r, B ) += 2.0 * J;
123  lhsd.var( r, C ) += 2.0 * J;
124  lhsd.var( r, D ) += 2.0 * J;
125 
126  const auto s = lhso.cptr( c, m_offset );
127  lhso.var( s, spidx(A,B) ) += J;
128  lhso.var( s, spidx(A,C) ) += J;
129  lhso.var( s, spidx(A,D) ) += J;
130 
131  lhso.var( s, spidx(B,A) ) += J;
132  lhso.var( s, spidx(B,C) ) += J;
133  lhso.var( s, spidx(B,D) ) += J;
134 
135  lhso.var( s, spidx(C,A) ) += J;
136  lhso.var( s, spidx(C,B) ) += J;
137  lhso.var( s, spidx(C,D) ) += J;
138 
139  lhso.var( s, spidx(D,A) ) += J;
140  lhso.var( s, spidx(D,B) ) += J;
141  lhso.var( s, spidx(D,C) ) += J;
142  }
143  }

146  //! Compute right hand side
147  //! \param[in] mult Multiplier differentiating the different stages in
148  //! multi-stage time stepping
149  //! \param[in] dt Size of time step
150  //! \param[in] coord Mesh node coordinates
151  //! \param[in] inpoel Mesh element connectivity
152  //! \param[in] U Solution vector at recent time step stage
153  //! \param[in] Un Solution vector at previous time step
154  //! \param[in,out] R Right-hand side vector computed
155  //! \author J. Bakosi
156  void rhs( tk::real mult,
157  tk::real dt,
158  const std::array< std::vector< tk::real >, 3 >& coord,
159  const std::vector< std::size_t >& inpoel,
160  const tk::MeshNodes& U,
161  const tk::MeshNodes& Un,
162  tk::MeshNodes& R )
163  {
164  Assert( U.nunk() == coord[0].size(), "Number of unknowns in solution "
165  "vector at recent time step incorrect" );
166  Assert( Un.nunk() == coord[0].size(), "Number of unknowns in solution "
167  "vector at previous time step incorrect" );
168  Assert( R.nunk() == coord[0].size(), "Number of unknowns in right-hand "
169  "side vector incorrect" );
170 
171  const auto& x = coord[0];
172  const auto& y = coord[1];
173  const auto& z = coord[2];
174 
175  // get reference to diffusivities for all components
176  const auto& diff =
177  g_inputdeck.get< tag::param, tag::advdiff, tag::diffusivity >().at(m_c);
178 
179  // zero right hand side for all components
180  for (ncomp_t c=0; c<m_ncomp; ++c) R.fill( c, m_offset, 0.0 );
181 
182  for (std::size_t e=0; e<inpoel.size()/4; ++e) {
183  const auto A = inpoel[e*4+0];
184  const auto B = inpoel[e*4+1];
185  const auto C = inpoel[e*4+2];
186  const auto D = inpoel[e*4+3];
187 
188  // compute element Jacobi determinant
189  const std::array< tk::real, 3 > ba{{ x[B]-x[A], y[B]-y[A], z[B]-z[A] }},
190  ca{{ x[C]-x[A], y[C]-y[A], z[C]-z[A] }},
191  da{{ x[D]-x[A], y[D]-y[A], z[D]-z[A] }};
192  const auto J = tk::triple( ba, ca, da );
193 
194  // construct tetrahedron element-level matrices
195 
196  // consistent mass
197  std::array< std::array< tk::real, 4 >, 4 > mass; // nnode*nnode [4][4]
198  // diagonal
199  mass[0][0] = mass[1][1] = mass[2][2] = mass[3][3] = J/60.0;
200  // off-diagonal
201  mass[0][1] = mass[0][2] = mass[0][3] =
202  mass[1][0] = mass[1][2] = mass[1][3] =
203  mass[2][0] = mass[2][1] = mass[2][3] =
204  mass[3][0] = mass[3][1] = mass[3][2] = J/120.0;
205 
206  // shape function derivatives
207  std::array< std::array< tk::real, 3 >, 4 > grad; // nnode*ndim [4][3]
208  grad[1] = tk::crossdiv( ca, da, J );
209  grad[2] = tk::crossdiv( da, ba, J );
210  grad[3] = tk::crossdiv( ba, ca, J );
211  for (std::size_t i=0; i<3; ++i)
212  grad[0][i] = -grad[1][i]-grad[2][i]-grad[3][i];
213 
214  // access solution at element nodes at time n
215  std::vector< std::array< tk::real, 4 > > u( m_ncomp );
216  for (ncomp_t c=0; c<m_ncomp; ++c)
217  u[c] = Un.extract( c, m_offset, A, B, C, D );
218  // access solution at element nodes at recent time step stage
219  std::vector< std::array< tk::real, 4 > > s( m_ncomp );
220  for (ncomp_t c=0; c<m_ncomp; ++c)
221  s[c] = U.extract( c, m_offset, A, B, C, D );
222  // access pointer to right hand side at component and offset
223  std::vector< const tk::real* > r( m_ncomp );
224  for (ncomp_t c=0; c<m_ncomp; ++c) r[c] = R.cptr( c, m_offset );
225 
226  // get velocity for problem
227  const auto vel =
228  Problem::template
229  velocity< tag::advdiff >
230  ( g_inputdeck, A, B, C, D, coord, m_c, m_ncomp );
231 
232  // add mass contribution to right hand side
233  for (ncomp_t c=0; c<m_ncomp; ++c)
234  for (std::size_t j=0; j<4; ++j) {
235  R.var(r[c],A) += mass[0][j] * u[c][j];
236  R.var(r[c],B) += mass[1][j] * u[c][j];
237  R.var(r[c],C) += mass[2][j] * u[c][j];
238  R.var(r[c],D) += mass[3][j] * u[c][j];
239  }
240 
241  // add advection contribution to right hand side
242  for (ncomp_t c=0; c<m_ncomp; ++c)
243  for (std::size_t j=0; j<4; ++j)
244  for (std::size_t k=0; k<3; ++k)
245  for (std::size_t l=0; l<4; ++l) {
246  R.var(r[c],A) -= mult * dt * mass[0][j] * vel[c][k][j]
247  * grad[l][k] * s[c][l];
248  R.var(r[c],B) -= mult * dt * mass[1][j] * vel[c][k][j]
249  * grad[l][k] * s[c][l];
250  R.var(r[c],C) -= mult * dt * mass[2][j] * vel[c][k][j]
251  * grad[l][k] * s[c][l];
252  R.var(r[c],D) -= mult * dt * mass[3][j] * vel[c][k][j]
253  * grad[l][k] * s[c][l];
254  }
255 
256  // add diffusion contribution to right hand side
257  for (ncomp_t c=0; c<m_ncomp; ++c)
258  for (std::size_t j=0; j<4; ++j)
259  for (std::size_t k=0; k<3; ++k) {
260  R.var(r[c],A) -= mult * dt * diff[c] * J * grad[0][k]
261  * grad[j][k] * s[c][j];
262  R.var(r[c],B) -= mult * dt * diff[c] * J * grad[1][k]
263  * grad[j][k] * s[c][j];
264  R.var(r[c],C) -= mult * dt * diff[c] * J * grad[2][k]
265  * grad[j][k] * s[c][j];
266  R.var(r[c],D) -= mult * dt * diff[c] * J * grad[3][k]
267  * grad[j][k] * s[c][j];
268  }
269  }
270  }
