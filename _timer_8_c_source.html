55 //******************************************************************************
56 {
57  using std::chrono::duration_cast;
58 
59  Dsec elapsed, estimated;
60 
61  if (it == 0) {
62 
63  // First iteration, just return zero
64  elapsed = estimated = clock::duration::zero();
65 
66  } else {
67 
68  // Compute time difference between start and now in seconds
69  elapsed = clock::now() - m_start;
70 
71  // Estimate time until term in seconds
72  Dsec est_term = elapsed * (term-time) / time;
73  // Estimate time until nstep in seconds
74  Dsec est_nstep = elapsed * static_cast<tk::real>(nstep-it) / it;

76  // Time stepping will stop at term or nstep, whichever is sooner
77  estimated = min(est_term, est_nstep);
