142 void
143 ExodusIIMeshWriter::writeElemBlock( int& elclass,
144  int64_t nnpe,
145  const std::string& eltype,
146  const std::vector< std::size_t >& inpoel )
147 const
148 //******************************************************************************
149 // Write element block to ExodusII file
150 //! \param[inout] elclass Count element class ids in file
151 //! \param[in] nnpe Number of nodes per element for block
152 //! \param[in] eltype String describing element type
153 //! \param[in] inpoel Element connectivity.
154 //! \author J. Bakosi
155 //******************************************************************************
156 {
157  if (inpoel.empty()) return;
158 
159  // increase number of element classes in file
160  ++elclass;
161 
162  // Make sure element connectivity starts with zero
163  Assert( *std::minmax_element( begin(inpoel), end(inpoel) ).first == 0,
164  "node ids should start from zero" );
165 
166  // Write element block information
167  ErrChk(
168  ex_put_elem_block( m_outFile,
169  elclass,
170  eltype.c_str(),
171  static_cast< int64_t >( inpoel.size() ) / nnpe,
172  nnpe,
173  0 ) == 0,
174  "Failed to write " + eltype + " element block to ExodusII file: " +
175  m_filename );
176 
177  // Write element connectivity with 1-based element ids
178  std::vector< int > inp;
179  for (auto p : inpoel) inp.push_back( static_cast< int >( p+1 ) );
180  ErrChk( ex_put_elem_conn( m_outFile, elclass, inp.data() ) == 0,
181  "Failed to write " + eltype + " element connectivity to ExodusII "
182  "file: " + m_filename );
183 }
