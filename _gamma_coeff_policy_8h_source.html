17  tk::ctr::ncomp_type ncomp,
18  const std::vector< kw::sde_b::info::expect::type >& b_,
19  const std::vector< kw::sde_S::info::expect::type >& S_,

22  std::vector< kw::sde_S::info::expect::type >& S,
23  std::vector< kw::sde_kappa::info::expect::type >& k )
24  \endcode
25  where
26  - ncomp denotes the number of scalar components of the system of gamma
27  SDEs.
28  - Constant references to b_, S_, and k_, which denote three vectors of
29  real values used to initialize the parameter vectors of the system of
30  gamma SDEs. The length of the vectors must be equal to the number of
31  components given by ncomp.
32  - References to b, S, and k, which denote the parameter vectors to be
33  initialized based on b_, S_, and k_.
34 
35  - Must define the static function _type()_, returning the enum value of the
36  policy option. Example:
37  \code{.cpp}
38  static ctrCoeffPolicyType type() noexcept {
39  return ctrCoeffPolicyType::CONSTANT;
40  }
41  \endcode
42  which returns the enum value of the option from the underlying option
43  class, collecting all possible options for coefficients policies.
