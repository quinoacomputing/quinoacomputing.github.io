2 /*!
3  \file src/DiffEq/GeneralizedDirichlet.h
4  \author J. Bakosi
5  \copyright 2012-2015, Jozsef Bakosi, 2016, Los Alamos National Security, LLC.
6  \brief Lochner's generalized Dirichlet SDE
7  \details This file implements the time integration of a system of stochastic
8  differential equations (SDEs) whose invariant is Lochner's [generalized
9  Dirichlet distribution](http://en.wikipedia.org/wiki/Generalized_Dirichlet_distribution).
10 
11  In a nutshell, the equation integrated governs a set of scalars, \f$0 \le
12  Y_i\f$, \f$i=1,\dots,K\f$, \f$\sum_{i=1}^KY_i\le1\f$, as
13  \f[ \begin{split}
14  \mathrm{d}Y_i(t) = \frac{\mathcal{U}_i}{2}\left\{ b_i\Big[S_i
15  \mathcal{Y}_K - (1-S_i)Y_i\Big] + Y_i\mathcal{Y}_K
16  \sum_{j=i}^{K-1}\frac{c_{ij}}{\mathcal{Y}_j}\right\}\mathrm{d}t +
17  \sqrt{\kappa_i Y_i \mathcal{Y}_K \mathcal{U}_i}\mathrm{d}W_i(t), \\
18  \qquad i=1,\dots,K, \end{split}
19  \f]
20  where \f$\mathrm{d}W_i(t)\f$ is an isotropic vector-valued [Wiener
21  process](http://en.wikipedia.org/wiki/Wiener_process) with independent
22  increments. The statistically stationary solution of the above coupled
23  system of nonlinear stochastic differential equations is the generalized
24  Dirichlet distribution,
25  \f[
26  \newcommand{\bv}[1]{{\mbox{$\mathbf{#1}$}}}
27  \mathscr{G}(\bv{Y},\bv{\alpha},\bv{\beta}) =
28  \prod_{i=1}^K\frac{\Gamma(\alpha_i+\beta_i)}{\Gamma(\alpha_i)
29  \Gamma(\beta_i)}Y_i^{\alpha_i-1} \mathcal{Y}_i^{\gamma_i} \qquad
30  \mathrm{with} \qquad \mathcal{Y}_i = 1-\sum_{k=1}^i Y_k,
31  \f]
32  provided the coefficients, \f$b_i\!>\!0\f$, \f$\kappa_i\!>\!0\f$,
33  \f$0\!<\!S_i\!<\!1\f$, and \f$c_{ij}\f$, with \f$c_{ij}\!=\!0\f$ for
34  \f$i\!>\!j\f$, \f$i,j\!=\!1,\dots,K\!-\!1\f$, satisfy
35  \f[ \begin{split}
36  \alpha_i & = \frac{b_i}{\kappa_i}S_i, \qquad i=1,\dots,K,\\
37  1-\gamma_i & = \frac{c_{1i}}{\kappa_1} = \dots = \frac{c_{ii}}{\kappa_i},
38  \qquad i=1,\dots,K-1,\\
39  1+\gamma_K & = \frac{b_1}{\kappa_1}(1-S_1) = \dots =
40  \frac{b_K}{\kappa_K}(1-S_K). \end{split}
41  \f]
42 
43  Here \f$\mathcal{U}_i = \prod_{j=1}^{K-i}\mathcal{Y}_{K-j}^{-1}\f$,
44  \f$\alpha_i>0\f$, and \f$\beta_i>0\f$ are parameters, while
45  \f$\gamma_i=\beta_i-\alpha_{i+1}-\beta_{i+1}\f$ for \f$i=1,\dots,K-1\f$, and
46  \f$\gamma_K=\beta_K-1\f$. \f$\Gamma(\cdot)\f$ denotes the [gamma
47  function](http://en.wikipedia.org/wiki/Gamma_function). To keep the
48  invariant distribution generalized Dirichlet, the above set of constraints
49  on the coefficients must be satisfied. For more details on the generalized
50  Dirichlet SDE, see http://dx.doi.org/10.1063/1.4822416.
51 */

118  //! \brief Advance particles according to the generalized Dirichlet SDE
119  //! \param[in,out] particles Array of particle properties
120  //! \param[in] stream Thread (or more precisely stream) ID
121  //! \param[in] dt Time step size
122  //! \author J. Bakosi
123  void advance( tk::Particles& particles,
124  int stream,
125  tk::real dt,
126  tk::real,
127  const std::map< tk::ctr::Product, tk::real >& )
128  {
129  const auto npar = particles.nunk();
130  for (auto p=decltype(npar){0}; p<npar; ++p) {
131  // Y_i = 1 - sum_{k=1}^{i} y_k
132  std::vector< tk::real > Y( m_ncomp );
133  Y[0] = 1.0 - particles( p, 0, m_offset );
134  for (ncomp_t i=1; i<m_ncomp; ++i)
135  Y[i] = Y[i-1] - particles( p, i, m_offset );
136 
137  // U_i = prod_{j=1}^{K-i} 1/Y_{K-j}
138  std::vector< tk::real > U( m_ncomp );
139  U[m_ncomp-1] = 1.0;
140  for (long i=static_cast<long>(m_ncomp)-2; i>=0; --i) {
141  auto I = static_cast< std::size_t >( i );
142  U[I] = U[I+1]/Y[I];
143  }
144 
145  // Generate Gaussian random numbers with zero mean and unit variance
146  std::vector< tk::real > dW( m_ncomp );
147  m_rng.gaussian( stream, m_ncomp, dW.data() );
148 
149  // Advance first m_ncomp (K=N-1) scalars
150  ncomp_t k=0;
151  for (ncomp_t i=0; i<m_ncomp; ++i) {
152  tk::real& par = particles( p, i, m_offset );
153  tk::real d = m_k[i] * par * Y[m_ncomp-1] * U[i] * dt;
154  d = (d > 0.0 ? sqrt(d) : 0.0);
155  tk::real a=0.0;
156  for (ncomp_t j=i; j<m_ncomp-1; ++j) a += m_cij[k++]/Y[j];
157  par += U[i]/2.0*( m_b[i]*( m_S[i]*Y[m_ncomp-1] - (1.0-m_S[i])*par ) +
158  par*Y[m_ncomp-1]*a )*dt + d*dW[i];
159  }
160  }
161  }
