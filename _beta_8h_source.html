2 /*!
3  \file src/DiffEq/Beta.h
4  \author J. Bakosi
5  \date Mon 01 Jun 2015 02:38:08 PM MDT
6  \copyright 2012-2015, Jozsef Bakosi.
7  \brief System of beta SDEs
8  \details This file implements the time integration of a system of stochastic
9  differential equations (SDEs) with linear drift and quadratic diagonal
10  diffusion, whose invariant is the joint [beta
11  distribution](http://en.wikipedia.org/wiki/Beta_distribution).
12 
13  In a nutshell, the equation integrated governs a set of scalars,
14  \f$0\!\le\!Y_\alpha\f$, \f$\alpha\!=\!1,\dots,N\f$, as
15  \f[
16  \mathrm{d}Y_\alpha(t) = \frac{b_\alpha}{2}\left(S_\alpha - Y_\alpha\right)
17  \mathrm{d}t + \sqrt{\kappa_\alpha Y_\alpha(1-Y_\alpha)}
18  \mathrm{d}W_\alpha(t), \qquad \alpha=1,\dots,N
19  \f]
20  with parameter vectors \f$b_\alpha > 0\f$, \f$\kappa_\alpha > 0\f$, and \f$0
21  < S_\alpha < 1\f$. Here
22  \f$\mathrm{d}W_\alpha(t)\f$ is an isotropic vector-valued [Wiener
23  process](http://en.wikipedia.org/wiki/Wiener_process) with independent
24  increments. The invariant distribution is the joint beta distribution. This
25  system of SDEs consists of N independent equations. For
26  more on the beta SDE, see http://dx.doi.org/10.1080/14685248.2010.510843.
27 */

86  //! \brief Advance particles according to the system of beta SDEs
87  //! \param[inout] particles Array of particle properties
88  //! \param[in] stream Thread (or more precisely stream) ID
89  //! \param[in] dt Time step size
90  //! \param[in] moments Map of statistical moments
91  //! \author J. Bakosi
92  void advance( tk::ParProps& particles,
93  int stream,
94  tk::real dt,
95  const std::map< tk::ctr::Product, tk::real >& moments )
96  {
97  const auto npar = particles.npar();
98  for (auto p=decltype(npar){0}; p<npar; ++p) {
99  // Generate Gaussian random numbers with zero mean and unit variance
100  tk::real dW[m_ncomp];
101  m_rng.gaussian( stream, m_ncomp, dW );
102 
103  // Advance all m_ncomp scalars
104  for (ncomp_t i=0; i<m_ncomp; ++i) {
105  tk::real& par = particles( p, i, m_offset );
106  tk::real d = m_k[i] * par * (1.0 - par) * dt;
107  d = (d > 0.0 ? std::sqrt(d) : 0.0);
108  par += 0.5*m_b[i]*(m_S[i] - par)*dt + d*dW[i];
109  }
110  }
111  }
