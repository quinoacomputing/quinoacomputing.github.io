62  //! \param[in] ncomp Number of components in this PDE
63  //! \param[in] mult Multiplier differentiating the different stages in

72  static void
73  diffusionRhs( tk::ctr::ncomp_type e,
74  tk::ctr::ncomp_type ncomp,
75  tk::real mult,
76  tk::real dt,
77  tk::real J,
78  const std::array< std::size_t, 4 >& N,
79  const std::array< std::array< tk::real, 3 >, 4 >& grad,
80  const std::vector< std::array< tk::real, 4 > >& s,
81  const std::vector< const tk::real* >& r,
82  tk::MeshNodes& R )
83  {
84  // get reference to diffusivities for all components
85  const auto& diff =
86  g_inputdeck.get< tag::param, tag::transport, tag::diffusivity >().at(e);
87 
88  // add diffusion contribution to right hand side
89  for (ncomp_t c=0; c<ncomp; ++c) {
90  tk::real a = mult * dt * diff[c] * J;
91  for (std::size_t j=0; j<4; ++j)
92  for (std::size_t k=0; k<3; ++k) {
93  R.var(r[c],N[0]) -= a * grad[0][k] * grad[j][k] * s[c][j];
94  R.var(r[c],N[1]) -= a * grad[1][k] * grad[j][k] * s[c][j];
95  R.var(r[c],N[2]) -= a * grad[2][k] * grad[j][k] * s[c][j];
96  R.var(r[c],N[3]) -= a * grad[3][k] * grad[j][k] * s[c][j];
97  }
98  }
99  }
100 
101  static ctr::PhysicsType type() noexcept
102  { return ctr::PhysicsType::ADVDIFF; }
103 };
