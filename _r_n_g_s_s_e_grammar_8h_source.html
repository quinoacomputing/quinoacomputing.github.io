21  //! \brief convert and insert sequence option value to map at position given
22  //! by tags
23  //! \author J. Bakosi
24  template< class Stack, template < class > class use, class Option,
25  typename field, typename sel, typename vec, typename tag,
26  typename... tags >
27  struct insert_seq : pegtl::action_base<
28  insert_seq< Stack, use, Option, field, sel, vec, tag,
29  tags... > > {
30  static void apply( const std::string& value, Stack& stack ) {
31  ctr::RNGSSESeqLen opt;
32  using EnumType = ctr::RNGSSESeqLen::EnumType;
33  // get recently inserted key from <sel,vec>
34  using key_type =
35  typename Stack::template nT< sel >::template nT< vec >::value_type;
36  const key_type& key = stack.template get< sel, vec >().back();
37  // Error out if RNG does not support option specified
38  if ( !ctr::RNG().supportsOpt( key, opt.value(value) ) ) {
39  grm::Message< Stack, grm::ERROR, grm::MsgKey::UNSUPPORTED >
40  ( stack, value );
41  }
42  stack.template insert_opt< key_type, field, EnumType, tag, tags... >
43  ( key, opt.value(value) );
44  // trigger error at compile-time if any of the expected option values
45  // is not in the keywords pool of the grammar
46  boost::mpl::for_each< typename Option::keywords >
47  ( tk::grm::is_keyword< use >() );
48  }
49  };
