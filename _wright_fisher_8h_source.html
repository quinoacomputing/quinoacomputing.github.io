111  //! \brief Advance particles according to the Wright-Fisher SDE
112  //! \param[inout] particles Array of particle properties
113  //! \param[in] stream Thread (or more precisely stream) ID
114  //! \param[in] dt Time step size
115  //! \param[in] t Physical time
116  //! \param[in] moments Map of statistical moments
117  //! \author J. Bakosi
118  void advance( tk::Particles& particles,
119  int stream,
120  tk::real dt,
121  tk::real t,
122  const std::map< tk::ctr::Product, tk::real >& moments )
123  {
124  // Compute sum of coefficients
125  const auto omega = std::accumulate( begin(m_omega), end(m_omega), 0.0 );
126  const auto npar = particles.nunk();
127 
128  for (auto p=decltype(npar){0}; p<npar; ++p) {
129  // Need to build the square-root of the Wright-Fisher diffusion matrix:
130  // B_ij = y_i * ( delta_ij - y_j ). If the matrix is positive definite,
131  // the Cholesky decomposition would work, however, B_ij is only positive
132  // semi-definite, so Cholesky may fail and considering floating-point
133  // errors it will definitely fail at some point.
134  //
135  // A stable square-root for B_ij is not yet implemented. Here is some
136  // details and points for further development:
137  //
138  // An interesting fact from http://math.stackexchange.com/a/332465, on
139  // gauging the eigen values of a matrix:
140  //
141  // "Choosing each diagonal entry to be greater than the sum of the
142  // absolute values of the other entries in the same row will immediately
143  // imply that all of the eigenvalues of A are positive, and therefore
144  // that A is positive definite."
145  //
146  // The WF diffusion matrix seems to be positive semi-definite with at
147  // least one very small eigenvalue O(10e-18). The row and column sums
148  // are zero. It does not appear that the matrix has negative
149  // eigenvalues.
150  //
151  // Using Cholesky decomposition to compute the square-root may fail.
152  // Other methods, such as diagonalizing the matrix or the LDL
153  // decomposition, may work, but have not tried. The former requires
154  // finding the eigenvalues, with e.g., LAPACK's DSPEVD, thereby
155  // diagonalizing the matrix, and taking the square-root of the diagonal
156  // elements to find the square-root of the matrix. See
157  // http://en.wikipedia.org/wiki/Square_root_of_a_matrix. This may be
158  // okay, but probably overkill, as we don't really need the eigenvalues,
159  // only the square-root of the matrix.
160  //
161  // The latter (LDL) is similar to Cholesky, but does not involve square
162  // root. See http://en.wikipedia.org/wiki/Cholesky_decomposition. See
163  // also Golub van Loan: Sec.4.1.2 LDL. It appears that LDL would compare
164  // in perfromance to Cholesky and would work even on large indefinite
165  // matrices. LAPACK probably has LDL.
166  //
167  // I'm putting this aside for now. The next step should probably be
168  // idintifying the LAPACK for LDL and use it to advance the WF system
169  // using LD^{1/2}.
170  //
171  // See also:
172  // http://en.wikipedia.org/wiki/Positive-definite_matrix
173  // file:///opt/intel/composerxe/Documentation/en_US/mkl/mklman/index.htm
174  // http://icl.cs.utk.edu/lapack-forum/archives/lapack/msg01497.html
175  // http://yarchive.net/comp/sqrtm.html
176 
177  tk::real B[m_ncomp][m_ncomp];
178  tk::real Bo[m_ncomp][m_ncomp];
179  for (ncomp_t i=0; i<m_ncomp; ++i) {
180  const tk::real& pari = particles( p, i, m_offset );
181  for (ncomp_t j=0; j<m_ncomp; ++j) {
182  const tk::real& parj = particles( p, j, m_offset );
183  if (i == j) {
184  B[i][i] = std::abs( pari * (1.0 - pari) );
185  if (B[i][i] < 1.0e-10) B[i][i] = 1.0;
186  } else {
187  B[i][j] = -pari*parj;
188  }
189  }
190  }
191  std::memcpy( Bo, B, m_ncomp*m_ncomp*sizeof(tk::real) );
192  // Compute diffusion matrix (lower triangle of Cholesky-decomposition)
193  lapack_int n = static_cast< lapack_int >( m_ncomp );
194  lapack_int info = LAPACKE_dpotrf( LAPACK_ROW_MAJOR, 'L', n, B[0], n );
195  if (info != 0 ) {
196  print_matrix( "=======\nOriginal Matrix", Bo[0], n, n );
197  std::cout << "info: " << info << std::endl;
198  print_matrix( "Result of Cholesky factorization", B[0], n, n );
199  for (ncomp_t i=0; i<m_ncomp; ++i) {
200  std::cout <<
201  std::setprecision( std::numeric_limits< tk::real >::digits10 )
202  << i << " par: " << particles( p, i, m_offset) << std::endl;
203  }
204  }
205  //ErrChk( info == 0, "Wright-Fisher Cholesky decomposition unsuccessful, "
206  // "info = " + std::to_string( info ) + ", particle: " +
207  // std::to_string( p ) );
208 
209  // Advance the first m_ncomp (N-1) scalars
210  if (info == 0) {
211  ncomp_t i = 0;
212  for (i=0; i<m_ncomp-1; ++i) {
213  tk::real& par = particles( p, i, m_offset );
214  // Advance first m_ncomp (K=N-1) scalars due to drift
215  par += 0.5*(m_omega[i] - omega*par)*dt;
216  // Advance first m_ncomp (K=N-1) particles with Cholesky-decomposed
217  // lower triangle (diffusion matrix)
218  for (ncomp_t j=0; j<m_ncomp-1; ++j)
219  if (j<=i) {
220  tk::real dW;
221  m_rng.gaussian( stream, 1, &dW );
222  par += B[i][j] * sqrt(dt) * dW;
223  }
224  }
225  // Compute the (N-1)th scalar from unit-sum
226  tk::real& par = particles( p, i, m_offset );
227  par = 1.0 - particles( p, 0, m_offset );
228  for (i=1; i<m_ncomp-1; ++i) {
229  par -= particles( p, i, m_offset );
230  }
231  }
232  }
233  }
