74 {
75  if (!nfail && !nwarn && !nskip && !nexcp) {
76  print.note< tk::QUIET >
77  ( "All " + std::to_string(ncomplete) + " " + suite + " tests passed" );
78  } else {
79  std::string skip, warn, fail, excp;
80  if (nwarn) warn = "finished with a warning: " + std::to_string(nwarn);
81  if (nskip) skip = std::string(nwarn ? ", " : "") +
82  "skipped: " + std::to_string(nskip);
83  if (nexcp) excp = std::string(nskip || nwarn ? ", " : "") +
84  "threw exception: " + std::to_string(nexcp);
85  if (nfail) fail = std::string(nexcp || nskip || nwarn ?
86  ", " : "") + "failed: " + std::to_string(nfail);
87  print.note< tk::QUIET >
88  ( "Of " + std::to_string(ncomplete) + " " + suite +
89  " tests total: " + warn + skip + excp + fail );
90  }
91 
92  return nfail || nexcp ? false : true;
