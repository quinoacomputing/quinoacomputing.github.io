47 //! \brief Charm++ main chare for the mesh converter executable, meshconv.
48 //! \details Note that this object should not be in a namespace.
49 class Main : public CBase_Main {
50 
51  public:
52  //! \brief Constructor
53  //! \details The main chare constructor is the main entry point of the
54  //! program, called by the Charm++ runtime system. The constructor does
55  //! basic initialization steps, e.g., parser the command-line, prints out
56  //! some useful information to screen (in verbose mode), and instantiates
57  //! a driver. Since Charm++ is fully asynchronous, the constructor
58  //! usually spawns asynchronous objects and immediately exits. Thus in the
59  //! body of the main chare constructor we fire up an 'execute' chare,
60  //! which then calls back to Main::execute(). Finishing the main chare
61  //! constructor the Charm++ runtime system then starts the
62  //! network-migration of all global-scope data (if any). The execute chare
63  //! calling back to Main::execute() signals the end of the migration of
64  //! the global-scope data. Then we are ready to execute the driver which
65  //! calls back to Main::finalize() when it finished. Then finalize() exits
66  //! by calling Charm++'s CkExit(), shutting down the runtime system.
67  //! \see http://charm.cs.illinois.edu/manuals/html/charm++/manual.html
68  Main( CkArgMsg* msg )
69  try :
70  // Parse command line into m_cmdline using default simple pretty printer
71  m_cmdParser( msg->argc, msg->argv, tk::Print(), m_cmdline ),
72  // Create pretty printer initializing output streams based on command line
73  m_print( m_cmdline.get< tag::verbose >() ? std::cout : std::clog ),
74  // Create MeshConv driver
75  m_driver( tk::Main< meshconv::MeshConvDriver >
76  ( msg->argc, msg->argv,
77  m_cmdline,
78  tk::HeaderType::MESHCONV,
79  MESHCONV_EXECUTABLE,
80  m_print ) ),
81  m_timer(1) // Start new timer measuring the total runtime
82  {
83  delete msg;
84  mainProxy = thisProxy;
85  // Fire up an asynchronous execute object, which when created at some
86  // future point in time will call back to this->execute(). This is
87  // necessary so that this->execute() can access already migrated
88  // global-scope data.
89  CProxy_execute::ckNew();
90  // Start new timer measuring the migration of global-scope data
91  m_timer.emplace_back();
92  } catch (...) { tk::processExceptionCharm(); }
93 
94  void execute() {
95  try {
96  m_timestamp.emplace_back("Migrate global-scope data", m_timer[1].hms());
97  m_driver.execute();
98  } catch (...) { tk::processExceptionCharm(); }
99  }
100 
101  void finalize() {
102  try {
103  m_timestamp.emplace_back( "Total runtime", m_timer[0].hms() );
104  m_print.time( "Timers (h:m:s)", m_timestamp );
105  m_print.endpart();
106  } catch (...) { tk::processExceptionCharm(); }
107  // Tell the Charm++ runtime system to exit
108  CkExit();
109  }
