23 Reader::Reader( const std::string& filename ) : m_filename( filename )
24 //******************************************************************************
25 // Constructor: Acquire file handle
26 //! \param[in] filename Name of file to open for reading
27 //! \author J. Bakosi
28 //******************************************************************************
29 {
30  // Make sure there is a filename
31  Assert( !filename.empty(), "No filename specified" );
32 
33  // Check if file exists, throw exception if it does not
34  m_inFile.open( filename, std::ifstream::in );
35  ErrChk( m_inFile.good(), "Failed to open file: " + filename );
36 
37  // Attempt to read a character, throw if it fails
38  // It is curious that on some systems opening a directory instead of a file
39  // with the above ifstream::open() call does not set the failbit. Thus we get
40  // here fine, so we try to read a character from it. If it is a directory or
41  // an empty file the read will fail, so we throw. Read more at: http://
42  // stackoverflow.com/questions/9591036/
43  // ifstream-open-doesnt-set-error-bits-when-argument-is-a-directory.
44  m_inFile.get();
45  ErrChk( m_inFile.good(), "Failed to read from file: " + filename );
46 
47  // Close it
48  m_inFile.close();
49  ErrChk( !m_inFile.fail(), "Failed to close file: " + filename );
50 
51  // Re-open
52  m_inFile.open( filename, std::ifstream::in );
53  ErrChk( m_inFile.good(), "Failed to open file: " + filename );
54 }
