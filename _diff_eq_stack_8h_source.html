80  //! \brief Function object for registering a differential equation into the
81  //! differential equation factory
82  //! \details This functor is repeatedly called by MPL's cartesian_product,
83  //! sweeping all combinations of the differential equation policies. The
84  //! purpose of template template is to simplify client code as that will
85  //! not have to specify the template arguments of the template argument
86  //! (the policies of Eq), since we can figure it out here. See also
87  //! http://stackoverflow.com/a/214900
88  //! \author J. Bakosi
89  template< template< class, class > class Eq >
90  struct registerDiffEq {
91  //! Need to store the reference to factory we are registering into
92  DiffEqFactory& factory;
93  //! Need to store which differential equation we are registering
94  const ctr::DiffEqType type;
95  //! Constructor, also count number of unique equation types registered
96  explicit registerDiffEq( DiffEqFactory& f,
97  ctr::DiffEqType t,
98  std::set< ctr::DiffEqType >& eqTypes ) :
99  factory( f ), type( t ) { eqTypes.insert( t ); }
100  //! \brief Function call operator called by mpl::cartesian_product for
101  //! each unique sequence of policy combinations
102  template< typename U > void operator()( U ) {
103  namespace mpl = boost::mpl;
104  // Get Initialization policy: 1st type of mpl::vector U
105  using InitPolicy = typename mpl::at< U, mpl::int_<0> >::type;
106  // Get coefficients policy: 2nd type of mpl::vector U
107  using CoeffPolicy = typename mpl::at< U, mpl::int_<1> >::type;
108  // Build differential equation key
109  ctr::DiffEqKey key{ type, InitPolicy::type(), CoeffPolicy::type() };
110  // Register equation (with policies given by mpl::vector U) into factory
111  tk::recordModelLate< DiffEq, Eq< InitPolicy, CoeffPolicy > >
112  ( factory, key, static_cast<ncomp_t>(0) );
113  }
114  };
115 
116  //! \brief Instantiate a differential equation
117  //! \details The template argument, EqTag, is used to find the given
118  //! differential equation in the input deck, the hierarchical data filled
119  //! during control file parsing, containing user input.
120  //! \param[in] eq The unique differential equation key whose object to
121  //! instantiate.
122  //! \param[in,out] cnt Counter, a std::map, that counts all instantiated
123  //! differential equations by type.
124  //! \author J. Bakosi
125  template< class EqTag >
126  DiffEq createDiffEq( ctr::DiffEqType eq,
127  std::map< ctr::DiffEqType, ncomp_t >& cnt ) const {
128  auto c = ++cnt[ eq ]; // count eqs
129  --c; // used to index vectors starting with 0
130  if ( g_inputdeck.get< tag::component, EqTag >()[c] ) {
131  // re-create key and search for it
132  ctr::DiffEqKey key{ eq,
133  g_inputdeck.get< tag::param, EqTag, tag::initpolicy >()[c],
134  g_inputdeck.get< tag::param, EqTag, tag::coeffpolicy >()[c] };
135  const auto it = m_factory.find( key );
136  Assert( it != end( m_factory ), "Can't find eq in factory" );
137  return it->second( c ); // instantiate and return diff eq object
138  } else Throw ( "Can't create DiffEq with zero independent variables" );
139  }
